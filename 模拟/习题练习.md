###### 日期时间相关

```c++
int month[]={0,31,28,31,30,31,30,31,31,30,31,30,31}; //每一个月对应的日期
//判断闰年
bool leap(int year){
    return year%4==0&&year%100!=0||year%400==0;
}
//给定yyyy-m-d-hh:mm格式的字符串,返回对应的分钟
ll get(string s){
	int year = stoi(s.substr(0,4)),m=stoi(s.substr(5,2)),
	d=stoi(s.substr(8,2)),hh=stoi(s.substr(11,2)),mm=stoi(s.substr(14,2));
//	cout << year << " " << m << " " << d << " " << hh << " " << mm << endl;
	ll ans = 0;
    for(int i = 0;i < year;i++){
        int day = 365+ leap(i);
        ans += day*1440;//24*60
    }
    for(int i = 1;i < m;i++){
        ans += (month[i] + (leap(year)&&i==2))*1440;
    }
    ans += (d-1)*1440;
    ans += hh*60;
    ans += mm;
    return ans;
}
```
