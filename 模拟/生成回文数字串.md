//根据数字长度c生成回文串
for(int base = 1,hi = 10; base <= 1000; base = base*10,hi = hi*10){
    for(int j = 0;j < 2; j++){ //按照[base长度*2 - 1,2*base长度] 进行构造回文
        for(int i = base; i < hi; i++){
            int x = i, y = (j == 0 ? i/10 : i); //奇数
            for(; y >  0;  y/= 10) x = x*10 + y%10; //构造回文串
            //例如:base:12 -> 121
            // base:12 -> 122 -> 1221
            palindrome.add(x);
        }
    }
}
