### 基础

宽度优先搜索适用于解决等权边的最短路问题.如果一个最优性问题可以抽象为等边权问题,则可以用宽度优先搜索解决.

```python
from collections import deque

# 棋盘大小
N = 4

# 方向数组，用于计算相邻位置
directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]

# 将二维棋盘转换为字符串
def board_to_str(board):
    return ''.join(''.join(map(str, row)) for row in board)

# 将字符串转换回二维棋盘
def str_to_board(s):
    return [list(map(int, s[i:i+N])) for i in range(0, len(s), N)]

# 判断坐标是否在棋盘范围内
def is_valid(x, y):
    return 0 <= x < N and 0 <= y < N

# BFS 搜索
def bfs(start, target):
    # 队列：存储当前棋盘状态、路径和交换步骤
    queue = deque([(start, [])])
    visited = set()  # 记录访问过的状态
    visited.add(start)

    while queue:
        # 当前状态和路径
        current, path = queue.popleft()

        # 如果到达目标状态，返回结果
        if current == target:
            return path

        # 当前棋盘状态
        board = str_to_board(current)

        # 遍历所有格子
        for x in range(N):
            for y in range(N):
                # 遍历相邻格子
                for dx, dy in directions:
                    nx, ny = x + dx, y + dy

                    if is_valid(nx, ny):
                        # 交换两个格子，生成新状态
                        new_board = [row[:] for row in board]
                        new_board[x][y], new_board[nx][ny] = new_board[nx][ny], new_board[x][y]
                        new_state = board_to_str(new_board)

                        # 如果新状态未访问过，加入队列
                        if new_state not in visited:
                            visited.add(new_state)
                            queue.append((new_state, path + [(x+1, y+1, nx+1, ny+1)]))

    return None  # 无解

# 输入
initial_board = [
    list(map(int, input().strip())) for _ in range(4)
]
target_board = [
    list(map(int, input().strip())) for _ in range(4)
]

# 初始状态和目标状态
start = board_to_str(initial_board)
target = board_to_str(target_board)

# BFS 搜索最短路径
result = bfs(start, target)

# 输出结果
if result:
    print(len(result))
    for step in result:
        print(''.join(map(str, step)))
else:
    print("No solution")

```
