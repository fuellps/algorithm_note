### Prim算法

基于贪心的思想

- 一开始所有点都在 **圈外**，用dis数组表示某一点到**圈外邻点**的最小距离。
- 随机选择一个源点 **S**,初始化dis[s]=0,其他点为无穷大
- 进行n轮迭代，每次取未访问的圈外点距离最小的，去更新其他点。
- 如果最后有n个点的距离都不为inf,则说明图是连通的。

```c++
//naive algorithm
bool prim(int s){
    memset(dis,0x3f,sizeof dis);
    memset(vis,0,sizeof vis);
    dis[s] = 0;
    int cnt = 0,ans = 0;
    for(int i = 1;i <= n; i++){
        int u = -1;
        for(int j = 1;j <= n; j++){
            if(!vis[j] && (u==-1 || dis[j] < dis[u])) u = j;
        }
        ans += dis[u];
        vis[u] = 1;
        if(dis[u] != 0x3f3f3f3f) cnt++;
        for(int i = h[u];i;i = ne[i]){
            int v = e[i];
            if(dis[v] > w[i]) dis[v] = w[i];
        }
    }
    return cnt == n;//判断图是否连通
}

//heap-optimize
bool heap_prim(int s){
    memset(dis,0x3f,sizeof dis);
    memset(vis,0,sizeof vis);
    //pii -> pair<int,int>
    priority_queue<pii,vector<pii>,greater<pii>> pq;
    dis[s] = 0;
    pq.push({s,0});
    int cnt = 0,ans = 0;
    while(!pq.empty()){
        int u,cost;
        tie(u,cost) = pq.top(); pq.pop();
        if(vis[u]) continue;
        vis[u] = 1;
        ans += cost;
        if(dis[u] != 0x3f3f3f3f) cnt++;
        for(int i = h[u];i;i = ne[i]){
            int v = e[i];
            if(dis[v] > w[i]){
                dis[v] = w[i];
                pq.push({v,dis[v]});
            } 
        }
    }
    return cnt == n;//判断图是否连通
}
```
