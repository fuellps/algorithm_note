设数组长度为n,则用堆式存储节点的总节点数约为为 $2^{\lceil logn \rceil + 1}- 1 = 4n$,所以**线段树长度**开到**4n**即可.

#### 单点更新

```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 10;
ll t[N<<2],a[N]; //注意线段树大小要开4*N.
int n,m;

//构建线段树，注意这里的线段树是左闭右开区间[0,n)范围。如上图所示。
void build(int x,int lx,int rx){
    if(rx - lx == 1){
        t[x] = a[lx];
        return;
    }
    int mid = lx + rx >> 1;
    build(x*2 + 1,lx,mid);
    build(x*2+2,mid,rx);
    t[x] = t[x*2 + 1] + t[x*2 + 2];
}
//查询[l,r)的区间和
ll query(int lx,int rx,int x,int l,int r){
    if(rx <= l || lx >= r) return 0;
    if(l <= lx && rx <= r) return t[x];
    int mid = lx + rx >> 1;
    return query(lx,mid,x*2+1,l,r) + query(mid,rx,x*2 + 2,l,r);
}
//更新原数组中下标为p的数为v
void update(int lx,int rx,int x,int p,int v){
    if(rx - lx == 1){
        t[x] = v;
        return;
    }
    int mid = lx + rx >> 1;
    if(p < mid) update(lx,mid,x*2+1,p,v);
    else update(mid,rx,x*2+2,p,v);
    t[x] = t[x*2+1] + t[x*2 + 2];
}
void solve() {
    cin >> n >> m;
    for(int i = 0;i < n; i++) cin >> a[i];
    build(0,0,n);
    while(m--){
        int op,x,y; cin >> op >> x >> y;
        if(op&1) update(0,n,0,x,y);
        else cout<<query(0,n,0,x,y) << endl;
    }
}
```

以下以单点更新，求区间最小值及区间个数为例，展示线段树下标从1开始的模板。

```c++
#include<bits/stdc++.h>
#define F first
#define S second
#define pii pair<int,int>
using namespace std;
const int N = 1e5 + 100;
int a[N];
pii t[N << 2];
int n, m;
//辅助函数。合并两个结点，更新对应的最小值及个数。
pii merge(const pii &a, const pii &b) {
    if (a.F < b.F) return {a};
    else if (b.F < a.F) return {b};
    else return {a.F, a.S + b.S};
}
//构建线段树，根节点从1开始，区间为左闭右闭[1,n]。
void build(int lx, int rx, int x) {
    if (rx == lx) {
        t[x] = {a[lx], 1};
        return;
    }
    int mid = lx + rx >> 1;
    build(lx, mid, x << 1);
    build(mid + 1, rx, x << 1 | 1);
    t[x] = merge(t[x << 1], t[x << 1 | 1]);
}
//查询[l,r]区间的最小值及个数。
pii query(int lx, int rx, int x, int l, int r) {
    if (rx < l || lx > r) return {INT_MAX, 0}; //当前结点与查询的区间无交集
    if (l <= lx && rx <= r) return t[x]; //完全在区间内
    int mid = lx + rx >> 1;
    return merge(query(lx, mid, x << 1, l, r), query(mid + 1, rx, x << 1 | 1, l, r));
}
//单点更新。将原数组下标为i（从1开始）的值更新为v.
void update(int lx, int rx, int x, int i, int v) {
    if (rx == lx) {
        t[x] = {v, 1};
        return;
    }
    int mid = lx + rx >> 1;
    if (i <= mid) update(lx, mid, x << 1, i, v);
    else update(mid + 1, rx, x << 1 | 1, i, v);
    t[x] = merge(t[x << 1], t[x << 1 | 1]);
}

void solve() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    build(1, n, 1); //建树
    while (m--) {
        int op, x, y;
        cin >> op >> x >> y;
        if (op & 1) update(1, n, 1, x + 1, y); //单点更新
        else {
            auto p = query(1, n, 1, x + 1, y); //区间查询
            cout << p.F << " " << p.S << endl;
        }
    }
}
```

单点更新求最大子段和。

```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 100;
int a[N];
struct item {
    ll seg, pref, suf, sum;
};
item t[N << 2];
int n, m;

//辅助函数。合并两个结点，更新对应的最大子段，最大前缀，最大后缀，元素和。
item merge(const item &a, const item &b) {
    return {max({a.seg, b.seg, a.suf + b.pref}),
            max(a.pref, a.sum + b.pref),
            max(b.suf, b.sum + a.suf), a.sum + b.sum};
}
//处理区间只有一个值的情况
item single(int v){
    if(v > 0){
        return {v,v,v,v};
    }
    return {0,0,0,v};
}
void build(int lx, int rx, int x) {
    if (rx == lx) {
        t[x] = single(a[lx]);
        return;
    }
    int mid = lx + rx >> 1;
    build(lx, mid, x << 1);
    build(mid + 1, rx, x << 1 | 1);
    t[x] = merge(t[x << 1], t[x << 1 | 1]);
}

//单点更新。将原数组下标为i（从1开始）的值更新为v.
void update(int lx, int rx, int x, int i, int v) {
    if (rx == lx) {
        t[x] = single(v);
        return;
    }
    int mid = lx + rx >> 1;
    if (i <= mid) update(lx, mid, x << 1, i, v);
    else update(mid + 1, rx, x << 1 | 1, i, v);
    t[x] = merge(t[x << 1], t[x << 1 | 1]);
}

void solve() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    build(1, n, 1);
    cout << t[1].seg << endl;
    while (m--) {
        int  x, y;
        cin >> x >> y;
         update(1, n, 1, x + 1, y);
        cout << t[1].seg << endl;
    }
}
```



#### 区间更新

```java
int f[N<<2]; //存放区间的和
int d[N<<2]; //懒更新标记

//建树 对s[l,r]建树
void build(int s,int t,int i){
    if(s == t){ //递归边界
         f[i] = nums[t];
        return;
    }
    int mid = s + t >>> 1;
    build(s,mid,i<<1) ; //建立左子树
    build(mid + 1,t,i<<1 | 1);//建立右子树
    up(i); // f[i] = f[i<<1] + f[i<<1 | 1];
}

//查询区间范围之和
long getSum(int l,int r,int s,int t,int i){
    if(l <= s && t <= r) return s[i];
    int mid = s + t >>> 1;
    //没有完全包裹住.有懒标记需要传递下去
    if(d[i] > 0 ){
        down(i,mid - s + 1, t - mid);
    }
    long res = 0;
    //区间左范围有重叠
	if(l <= mid) res += getsum(l,r,s,mid,i<<1);
    //区间右范围有重叠
    if(r > mid) res += getsum(l,r,mid + 1,t,i<<1 | 1);
	return res;    
}

void update(int l,int r,int c,int s,int t,int i){
    if(l <= s && t <= r){
        s[i] += (s - t +1)*c;
        d[i] += c;
        return;
    }
    int mid = s + t >>> 1;
    if(d[i] > 0) down(i,mid -s + 1,r - mid,);
    //区间左范围有重叠
    if(l <= mid) update(l,r,c, s,mid, i<<1);
    //区间右范围有重叠
    if(r > mid) update(l,r,c,mid + 1,t, i << 1| 1);
    up[i]; //f[i] = f[i<<1] + f[i<<1  | 1];
}
    
 //向下传递懒更新信息
void down(int i,int nl,int nr){
    s[i<<1] += nl*d[i];
    s[i<<1 | 1] += nr*d[i];
    d[i<<1] += d[i];
    d[i<<1|1] |+= d[i];
    d[i] = 0;
}


//更新当前节点
void up(int i){
    s[i] = s[i<<1] + s[i<<1 | 1];
}

```

有几个优化的点:当前节点为叶子节点时,不需要向下传递懒更新信息.



### 以类的形式组织

```c++
struct Node {
    ll sum;
    ll add;
};

struct segtree {
    vector<Node> t;
    int n;

    segtree(int _n) : n(_n), t(_n << 2) {}

    void build(int lx, int rx, int x) {
        if (lx == rx) {
            t[x] = {a[lx]};
            return;
        }
        int mid = lx + rx >> 1;
        build(lx, mid, x << 1);
        build(mid + 1, rx, x << 1 | 1);
        t[x] = merge(t[x << 1], t[x << 1 | 1]);
    }

    void build(){
        build(1,n,1);
    }


    Node merge(const Node &a, const Node &b) {
        return {a.sum + b.sum, 0};
    }


    void propagate(int x, int lx, int rx) {
        int mid = lx + rx >> 1;
        int l = mid - lx + 1, r = rx - mid;
        if (t[x].add) {
            t[x << 1].add += t[x].add, t[x<<1].sum += l*t[x].add;
            t[x << 1 | 1].add += t[x].add,t[x<<1|1].sum += r*t[x].add;
            t[x].add =0 ;
        }
    }


    void update(int lx, int rx, int x, int l, int r, ll v) {
        if (lx > r || rx < l) return;
        if (l <= lx && rx <= r) {
            t[x].add += v;
            t[x].sum += (rx-lx+1)*v;
            return;
        }
        propagate(x, lx, rx);
        int mid = lx + rx >> 1;
        update(lx, mid, x << 1, l, r, v);
        update(mid + 1, rx, x << 1 | 1, l, r, v);
        t[x] = merge(t[x << 1], t[x << 1 | 1]);
    }


    ll query(int lx, int rx, int x, int l, int r) {
        if (lx > r || rx < l) return null;
        if (l <= lx && rx <= r) return t[x].sum;
        propagate(x, lx, rx);
        int mid = lx + rx >> 1;
        return query(lx, mid, x << 1, l, r) + query(mid + 1, rx, x << 1 | 1, l, r);
    }



    void modify(int lx, int rx, int x, int i, ll v) {
        if (lx == rx) {
            t[x].val += v;
            return;
        }
        int mid = lx + rx >> 1;
        if (i <= mid) {
            modify(lx, mid, x << 1, i, v);
        } else {
            modify(mid + 1, rx, x << 1 | 1, i, v);
        }
        t[x] = merge(t[x << 1], t[x << 1 | 1]);
    }


    void update(int l, int r, ll v) {
        update(1, n, 1, l, r, v);
    }

    void modify(int i, ll v) {
        modify(1, n, 1, i, v);
    }

    ll query(int l, int r) {
        return query(1, n, 1, l, r);
    }


};
```
