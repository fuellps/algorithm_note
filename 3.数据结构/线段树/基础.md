#### OP1修改操作：set(i,v)  将原数组下标为i的数修改为v

执行$set(2,3)$,则线段树修改如下：

> 可见，修改操作涉及到的结点只有 $\log n$个，因此复杂度为 $O(\log n)$

#### OP2区间查询：query(l,r) 查询[l,r-1]的区间和

![image-20250316114648440](基础.assets/image-20250316114648440.png)

- 结束递归有两种情况：
  1. 递归到的区间[lx,rx)不在查询区间[l,r)内。此时返回不合法的状态。
  2. 递归到的区间[lx,rx)完全在查询区间[l,r)上。此时返回线段树的当前结点。

#### 应用

线段树应用很广泛，只有满足直到所求东西的两半，能够合并成答案，都可以用线段树。

如乘法，gcd，位运算等。。

### 模板

教程可参考 [Segment Tree, part 1 - Codeforces](https://codeforces.com/edu/course/2/lesson/4/1)

设数组长度为n,则用堆式存储节点的总节点数约为为 $2^{\lceil logn \rceil + 1}- 1 = 4n$,所以**线段树长度**开到**4n**即可.

#### 单点更新

```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 10;
ll t[N<<2],a[N]; //注意线段树大小要开4*N.
int n,m;

//构建线段树，注意这里的线段树是左闭右开区间[0,n)范围。如上图所示。
void build(int x,int lx,int rx){
    if(rx - lx == 1){
        t[x] = a[lx];
        return;
    }
    int mid = lx + rx >> 1;
    build(x*2 + 1,lx,mid);
    build(x*2+2,mid,rx);
    t[x] = t[x*2 + 1] + t[x*2 + 2];
}
//查询[l,r)的区间和
ll query(int lx,int rx,int x,int l,int r){
    if(rx <= l || lx >= r) return 0;
    if(l <= lx && rx <= r) return t[x];
    int mid = lx + rx >> 1;
    return query(lx,mid,x*2+1,l,r) + query(mid,rx,x*2 + 2,l,r);
}
//更新原数组中下标为p的数为v
void update(int lx,int rx,int x,int p,int v){
    if(rx - lx == 1){
        t[x] = v;
        return;
    }
    int mid = lx + rx >> 1;
    if(p < mid) update(lx,mid,x*2+1,p,v);
    else update(mid,rx,x*2+2,p,v);
    t[x] = t[x*2+1] + t[x*2 + 2];
}
void solve() {
    cin >> n >> m;
    for(int i = 0;i < n; i++) cin >> a[i];
    build(0,0,n);
    while(m--){
        int op,x,y; cin >> op >> x >> y;
        if(op&1) update(0,n,0,x,y);
        else cout<<query(0,n,0,x,y) << endl;
    }
}
```

以下以单点更新，求区间最小值及区间个数为例，展示线段树下标从1开始的模板。

```c++
#include<bits/stdc++.h>
#define F first
#define S second
#define pii pair<int,int>
using namespace std;
const int N = 1e5 + 100;
int a[N];
pii t[N << 2];
int n, m;
//辅助函数。合并两个结点，更新对应的最小值及个数。
pii merge(const pii &a, const pii &b) {
    if (a.F < b.F) return {a};
    else if (b.F < a.F) return {b};
    else return {a.F, a.S + b.S};
}
//构建线段树，根节点从1开始，区间为左闭右闭[1,n]。
void build(int lx, int rx, int x) {
    if (rx == lx) {
        t[x] = {a[lx], 1};
        return;
    }
    int mid = lx + rx >> 1;
    build(lx, mid, x << 1);
    build(mid + 1, rx, x << 1 | 1);
    t[x] = merge(t[x << 1], t[x << 1 | 1]);
}
//查询[l,r]区间的最小值及个数。
pii query(int lx, int rx, int x, int l, int r) {
    if (rx < l || lx > r) return {INT_MAX, 0}; //当前结点与查询的区间无交集
    if (l <= lx && rx <= r) return t[x]; //完全在区间内
    int mid = lx + rx >> 1;
    return merge(query(lx, mid, x << 1, l, r), query(mid + 1, rx, x << 1 | 1, l, r));
}
//单点更新。将原数组下标为i（从1开始）的值更新为v.
void update(int lx, int rx, int x, int i, int v) {
    if (rx == lx) {
        t[x] = {v, 1};
        return;
    }
    int mid = lx + rx >> 1;
    if (i <= mid) update(lx, mid, x << 1, i, v);
    else update(mid + 1, rx, x << 1 | 1, i, v);
    t[x] = merge(t[x << 1], t[x << 1 | 1]);
}

void solve() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    build(1, n, 1); //建树
    while (m--) {
        int op, x, y;
        cin >> op >> x >> y;
        if (op & 1) update(1, n, 1, x + 1, y); //单点更新
        else {
            auto p = query(1, n, 1, x + 1, y); //区间查询
            cout << p.F << " " << p.S << endl;
        }
    }
}
```



#### 区间更新

```java
int f[N<<2]; //存放区间的和
int d[N<<2]; //懒更新标记

//建树 对s[l,r]建树
void build(int s,int t,int i){
    if(s == t){ //递归边界
         f[i] = nums[t];
        return;
    }
    int mid = s + t >>> 1;
    build(s,mid,i<<1) ; //建立左子树
    build(mid + 1,t,i<<1 | 1);//建立右子树
    up(i); // f[i] = f[i<<1] + f[i<<1 | 1];
}

//查询区间范围之和
long getSum(int l,int r,int s,int t,int i){
    if(l <= s && t <= r) return s[i];
    int mid = s + t >>> 1;
    //没有完全包裹住.有懒标记需要传递下去
    if(d[i] > 0 ){
        down(i,mid - s + 1, t - mid);
    }
    long res = 0;
    //区间左范围有重叠
	if(l <= mid) res += getsum(l,r,s,mid,i<<1);
    //区间右范围有重叠
    if(r > mid) res += getsum(l,r,mid + 1,t,i<<1 | 1);
	return res;    
}

void update(int l,int r,int c,int s,int t,int i){
    if(l <= s && t <= r){
        s[i] += (s - t +1)*c;
        d[i] += c;
        return;
    }
    int mid = s + t >>> 1;
    if(d[i] > 0) down(i,mid -s + 1,r - mid,);
    //区间左范围有重叠
    if(l <= mid) update(l,r,c, s,mid, i<<1);
    //区间右范围有重叠
    if(r > mid) update(l,r,c,mid + 1,t, i << 1| 1);
    up[i]; //f[i] = f[i<<1] + f[i<<1  | 1];
}
    
 //向下传递懒更新信息
void down(int i,int nl,int nr){
    s[i<<1] += nl*d[i];
    s[i<<1 | 1] += nr*d[i];
    d[i<<1] += d[i];
    d[i<<1|1] |+= d[i];
    d[i] = 0;
}


//更新当前节点
void up(int i){
    s[i] = s[i<<1] + s[i<<1 | 1];
}

```

有几个优化的点:当前节点为叶子节点时,不需要向下传递懒更新信息.
