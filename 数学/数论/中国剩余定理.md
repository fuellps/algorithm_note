### 定义

中国剩余定理用于求解如下形式的一元线性同余方程组(其中$m1,\dots,m_k两两互质$)
$$
\left\{ 
\begin{array}{c}
    x \equiv a_1 \pmod {n_1} \\ 
    x \equiv a_2 \pmod {n_k} \\
    \vdots \\
    x \equiv a_k \pmod {n_k}
\end{array}
\right.
$$

### 算法流程

1. 计算所有模数的积 mul；
2. 对于第i个方程: 
   1. 计算$m_i =\frac{mul}{n_i}$
   2. 计算$m在模n_i意义下的逆元m_i^{-1}$
   3. 计算 $c_i=m_i*m_i^{-1}(不要对n_i取模)$
3. 方程组在模n意义下的唯一解为: $x= \sum_{i=1}^{k} a_i c_i \pmod n$

> 证明：
> $$
> \begin{aligned}
> &需要证明对于任意i=1,2,\dots,k满足x \equiv a_i \pmod {n_i}\\
> &当i \ne j时，m_i \mod n_j \equiv 0\\
> x&= \sum_{i=1}^{k} a_i c_i \pmod {n_i}\\
> &=a_im_im_i^{-1} \pmod {n_i} \\
> &=a_i\\
> &即对于任意i=1，2，\dots,k,总是满足x \equiv a_i \pmod {n_i}
> \end{aligned}
> $$
>

```c++
typedef long long ll;
ll CRT(int k,ll* a,ll* n){
    ll mul = 1;
    for(int i = 0;i < k; i++) mul=mul*n[i];
  	ll ans = 0;
    for(int i  =0;i < k; i++){
        ll m = mul/n[i];
        exgcd(m,n); 
        ans = (ans + a*x%mul*m)%mul;
    }
    return (ans%mul + mul)%mul;
}
```



注意这个写法虽然简单但可能会被卡。

### 扩展：模数不互质的情况下

对于模数不互质的情况，将两个方程合并为一个方程，依次进行处理。

#### 两个方程的情况

$$
\left\{ 
\begin{array}{c}
    t \equiv a_1 \pmod {m_1} \\ 
    t \equiv a_2 \pmod {m_2} \\
\end{array}
\right.
$$

$$
\begin{aligned}
&\therefore  a_1 \pmod {m_1} \equiv  a_2 \pmod {m_2}\\
&\therefore  a1*+ m_1*x_1 = a_2 + m_2*x_2\\
&\therefore  m_1*x_1 -  m_2*x_2 = a_2 - a_1 \\
&由裴蜀定理定理知:当gcd(m1,m2) \not{|}\ (a_2-a_1)时无解。\\
&否则可得通解x_1 = x_0*\frac{a2-a1}{gcd(m_1,m_2)} +  \frac{m_2}{gcd(m_1,m_2)}*k\\
&(其中x_0为扩欧求出的x_1特解)\\
\therefore t &= a_1 + m_1*x_1\\
&=a_1 + m_1*(x_0*\frac{a2-a1}{gcd(m_1,m_2)} +  \frac{m_2}{gcd(m_1,m_2)}*k)\\
&=m_1*x_0*\frac{a2-a1}{gcd(m_1,m_2)} + a_1  + lcm(m_1,m_2)*k\\
&\therefore m_1*x_0*\frac{a2-a1}{gcd(m_1,m_2)} + a_1 \pmod {lcm(m_1,m_2)}

\end{aligned}
$$

#### 多个方程的情况

按照上面的步骤一一合并即可。

```c++
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll x,y,n;
ll exgcd(ll a, ll b){
    if(!b){
        x = 1,y = 0;
        return a;
    }
    ll d = exgcd(b,a%b);
    ll t = x;x = y; y = t - a/b*y;
    return d;
}
// 计算 a*b%p:解决乘法y
ll mul(ll a,ll b, ll p){
    ll res = 0;
    while(b){
        if(b&1) res = (res + a)%p;
        a = (a + a)%p;
        b >>= 1;
    }
    return res;
}

int main(){
    cin>>n;
    ll M,r; cin >> M >> r; 
    bool flag =false;
    for(int i =  1; i <  n; i++){
        ll m,a; cin >> m >> a;
        ll b = ((a - r)%m+m)%m; //求得同余方程得最小正整数
        ll d = exgcd(M,m);
        if(b%d) {flag = true;break;} //不满足条件，无解
        x = mul(x,b/d,m/d); //计算同余方程最小整数解
        r += x*M; 
        M *= m/d;
        r = (r%M+M)%M;
    }
    if(flag) cout << -1;
    else cout << r << endl;
}
```
