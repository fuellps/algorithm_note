### 定义



$$
\mu(n) =
\begin{cases}
1, & n = 1,\\
0, & \text{$n$ 含有平方质因子},\\
(-1)^s, & s\text{ 为 $n$ 的不同质因子个数}.
\end{cases}
$$



### 模板

```c++
int prime[N],mu[N],cnt;
bool st[N];

void get_mu(int n){
    mu[1] = 1; //特殊定义
    for(int i = 2;i <= n; i++){
        if(!st[i]){
            mu[i]= -1;prime[cnt++] = i;
        }
        for(int j = 0;prime[j] <= n/i;j++){
            int t = prime[j]*i;
            st[t] = true;
            if(i%prime[j]==0){
                mu[t] = 0;
            }else{
                mu[t] -= mu[i];
            }
        }
    }
}
```
