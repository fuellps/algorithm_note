### 消元法的核心

1. 把任意两行交换位置，不会影响答案
2. 把某一行乘上非零常数，不会影响答案
3. 把某一行加上另一行的整数倍，不会影响答案

### 模板

#### 高斯消元解线性方程组

1. 遍历所有系数列，每一列执行如下操作：
   1. 计算当前列的最大值所在行，与当前行交换。
   2. 如果当前列最大值已经为0，则跳过，进入下一列。
   3. 将当前行第一个数变为1
   4. 将当前行下的当前列全变为0
2. 结束后对行所在位置进行讨论：
   1. 如果r<n:
      1. $存在 i \in r,\dots,n,使得a[i][n] > eps,则无解$
      2. 否则有无穷多组解
   2. 否则有解，根据最后的解从后往前递推即可。



```c++
#include <bits/stdc++.h>
using namespace std;

const int N = 110;
double a[N][N],f[N];
int n ;
double eps = 1e-6;
int gauss(){
    int r,c;
    for(c = 0,r = 0;c < n; c++){
        int t = r;
        for(int j = r+ 1;j < n; j++){
            if(fabs(a[j][c]) > fabs(a[t][c])) t= j; //找到当前列得最大值的那一行
        }
        for(int j  =c;j <= n; j++) swap(a[r][j], a[t][j]); //最大值那一行与当前行进行交换
        if(fabs(a[r][c]) < eps) continue; //有一列为0，要么无解，要么无穷多组解
        for(int i =  n;i >= c;i--){
            a[r][i] /= a[r][c]; //将当前枚举到的行第一个数变为1
        }
        for(int i = r + 1; i < n; i++){ //将当前行下面的所有行的第c列第1个数变为0
            if(fabs(a[i][c]) > eps){//不为0才操作
                for(int j = n; j >= c;j--){
                    a[i][j] -= a[r][j]*a[i][c]; 
                }
            }
        }
        r++;
    }
    if(r < n){
        //枚举每一列方程对应的值
        for(int i = r;i < n; i++){
            if(fabs(a[i][n]) > eps) return -1; //无解
        }
        return 0; //无穷多组解
    }
    //计算答案
    for(int i = n-1;i >= 0; i--){
        for(int j = i + 1; j < n; j++){
            a[i][n] -= a[i][j]*a[j][n];
        }
    }
    return 1; //有解
}
int main(){
    cin >> n;
    for(int i = 0;i < n; i++) for(int j = 0;j <= n; j++) cin >> a[i][j];
    int res = gauss();
    if(res == -1) puts("No solution");
    else if(res == 0) puts("Infinite group solutions");
    else for(int i = 0;i < n; i++) printf("%.2f\n",a[i][n]);
}
```

#### 高斯消元解异或方程组

异或满足交换律和结合律，故可以按照高斯消元法逐步消元求解，有以下注意点：

1. 消元时使用异或消元而非加法消元
2. 消元时不需要乘除改变系数（系数均为0,1）

```c++
int gauss(){
    int r,c;
    for(c = 0,r = 0; c < n; c++){ //枚举每一列
        int t = r; 
        for(int j = t + 1; j < n; j++) 
            if(a[j][c]) t = j;
        for(int j = c;j <= n; j++) swap(a[t][j], a[r][j]); //交换行
        if(!a[r][c]) continue;//无唯一解
        for(int i = r + 1; i < n; i++){
            if(a[i][c]){
                for(int j = c;j <= n; j++) a[i][j] ^= a[r][j];
            }
        }
        r++;
    }
    if(r < n){
        for(int i = r; i < n; i++){
            if(a[i][n]) return -1; //无解
        }
        return 0; //无穷多组解
    }
    //消元
    for(int i = n - 1; i >= 0;i --){
        for(int j = i + 1; j < n; j++){
            if(a[i][j]) a[i][n]^= a[j][n];
        }
    }
    return 1; //有唯一解
}
```
