### 存储

高精度数字利用字符串存储,每一个字符表示数字的一个十进制位。

> 读入字符串时，数字最高位在字符串首（下标小的位置）。但是习惯上，下标最小的位置存放的是数字的 $最低位$，即存储反转的字符串。

### 高精度结构体

支持如下操作：下面操作运算支持的数只为**非负数**

1. 高精度加法：数据范围 $0 \le a \le 10^{500}, 0 \le b \le 10^{500}$
1. 高精度减法
2. 高精度乘单精度.单精度范围不能太大，一般最大可达1e5~1e6.
2. 高精度乘高精度 
2. 高精度除单精度.
2. 赋值操作：支持字符串赋值
2. 支持小于比较运算。

```c++
struct Bigint{
    int a[MAX_SIZE]{};
    int len;
    bool neg;
    Bigint(): len(0){}
    Bigint(const string& s){
        len = s.length();
        for(int i =0 ;i < len; i++){
            a[i] = s[len - i - 1] - '0';
        }
    }
    bool operator <(const Bigint& b) const{
		if(len != b.len) return len < b.len;
		for(int i =0;i < len; i++){
			if(a[i] != b.a[i]) return a[i] < b.a[i];
		}
		return false;
	}

    Bigint operator+ (const Bigint& b) const{
        Bigint res;
        res.len = max(len,b.len);
        int& len = res.len;auto& t = res.a;
        //模拟加法过程
        for(int i = 0;i < len; i++){
            t[i] += a[i] + b.a[i];
            if(t[i] > 9){
                t[i+1]++;
                t[i] %= 10;
            }
        }
        if(t[len]) len++; //处理进位
        return res;
    }
    
    Bigint operator-(const Bigint& b) const{
		Bigint res;
		auto& t = res.a;
		if(*this<b){
			res.len = b.len;
			res.neg = true;
			for(int i =0 ;i < res.len; i++){
				t[i] += b.a[i] - a[i];
				if(t[i] < 0){
					t[i] += 10;
					t[i+1]--;
				}
			}
		}else{
			res.len = len;
			for(int i =0 ;i < len; i++){
				t[i] += a[i] - b.a[i];
				if(t[i] < 0){
					t[i+1]--;
					t[i] += 10;
				}
			}
		}
		while(res.len > 1 && t[res.len-1] == 0) res.len--;
		return res;
	}

    //重载赋值操作，可以将字符串赋值给Bigint
    Bigint& operator=(const string& s){
        fill(a,a+len,0);
        len = s.size();
        for(int i = 0;i < len;i++){
            a[i] = s[len-i-1] - '0';
        }
        return *this;
    }

    Bigint operator*(int b) const{
        Bigint res;res.len = len; int& len = res.len;auto& t = res.a;
        //每一位直接乘上单精度处理进位即可。
        for(int i = 0;i < len;i++){
            t[i] += a[i]*b;
            if(t[i] > 9){
                t[i+1] += t[i]/10;
                t[i] %= 10;
            }
        }
        //处理结尾进位。
        while(t[len]){
            if(t[len] > 9){
                t[len+1] += t[len]/10;
                t[len]%=10;
            }
            len++;
        }
        //处理前导0，有可能乘以0
        while(len > 1&& t[len-1] == 0) --len;
        return res;
    }


    Bigint operator*(const Bigint& b) const{
        Bigint res;res.len = len + b.len; int& length = res.len;auto& t = res.a;
        //高精度乘高精度，按位相乘.
        for(int i = 0;i < len; i++){
            for(int j = 0;j < b.len; j++){
                t[i+j] += a[i]*b.a[j];
                if(t[i+j] > 9){
                    t[i+j+1] += t[i+j]/10;
                    t[i+j]%=10;
                }
            }
        }
        //处理前导0
        while(length > 1&& t[length-1] == 0) --length;
        return res;
    }
    
    Bigint operator/(const int b) const{
        Bigint res;res.len = len; int& len = res.len;auto& t = res.a;
        long long r = 0; //存余数，从高位往低位计算
        for(int i = len - 1; i >= 0;i--){
			r = r*10+a[i];
            if(r>=b){
                t[i] = r/b;
                r%=b;
            }
        }
        while(len > 1 && t[len-1] == 0) len--; 
        return res;
    }

    friend  ostream& operator<<(ostream& os,const Bigint& bi){
    	if(bi.neg) os << "-";
        for(int i = bi.len - 1;i;i--){
            os << bi.a[i];
        }
        os <<bi.a[0];
        return os;
    }
};
```
